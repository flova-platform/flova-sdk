function(add_flova_logging_test target)
  add_executable(${target} test_flova_logging.cpp)
  target_include_directories(${target} PRIVATE
    ${PROJECT_SOURCE_DIR}/packages/flova-embedded-sdk/include ${PROJECT_SOURCE_DIR}/packages/flova-arduino/include)
  add_test(NAME ${target} COMMAND ${target})
endfunction()
